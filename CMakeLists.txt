cmake_minimum_required(VERSION 3.16)
project(p3cg LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3" CACHE STRING "" FORCE)

option(P3_OPENMP  "OpenMP threads inside each MPI rank" ON)
option(P3_OFFLOAD "OpenMP target offload of the same kernels (nvhpc: -mp=gpu; validated on a Vanda GPU node)" OFF)
option(P3_NATIVE  "-march=native (only when compiling on the node type that runs the code)" OFF)
set(P3_OFFLOAD_FLAGS "-mp=gpu" CACHE STRING "compile+link flags for P3_OFFLOAD (nvhpc default)")
set(PLANT_HALO_BUG 0 CACHE STRING "0 = correct build; 1 = skip one halo face; 2 = shift one halo face (validation only)")
set(PLANT_GHOST_BUG 0 CACHE STRING "p3graph: 0 = correct build; 1 = one neighbour's ghosts never stored (validation only)")

find_package(MPI REQUIRED COMPONENTS CXX)

if(P3_OPENMP AND NOT P3_OFFLOAD)
  # Apple clang has no -fopenmp driver flag; point FindOpenMP at Homebrew's libomp.
  if(APPLE AND CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
    foreach(_p /opt/homebrew/opt/libomp /usr/local/opt/libomp)
      if(EXISTS "${_p}/include/omp.h" AND NOT OpenMP_CXX_FLAGS)
        set(OpenMP_CXX_FLAGS "-Xpreprocessor -fopenmp -I${_p}/include" CACHE STRING "" FORCE)
        set(OpenMP_CXX_LIB_NAMES "omp" CACHE STRING "" FORCE)
        set(OpenMP_omp_LIBRARY "${_p}/lib/libomp.dylib" CACHE FILEPATH "" FORCE)
      endif()
    endforeach()
  endif()
  find_package(OpenMP REQUIRED COMPONENTS CXX)
endif()

execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
                OUTPUT_VARIABLE P3_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE _git_rc)
if(NOT _git_rc EQUAL 0 OR P3_GIT_SHA STREQUAL "")
  # A cluster copy has no .git; cluster/ops/sync.sh writes the commit into SYNCED.
  set(P3_GIT_SHA "unknown")
  if(EXISTS "${CMAKE_SOURCE_DIR}/SYNCED")
    file(STRINGS "${CMAKE_SOURCE_DIR}/SYNCED" _synced REGEX "^commit ")
    if(_synced)
      string(SUBSTRING "${_synced}" 7 12 _c)
      set(P3_GIT_SHA "${_c}-synced")
    endif()
  endif()
else()
  execute_process(COMMAND git status --porcelain --untracked-files=no -- src CMakeLists.txt
                  WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} OUTPUT_VARIABLE _dirty OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT _dirty STREQUAL "")
    set(P3_GIT_SHA "${P3_GIT_SHA}-dirty")
  endif()
endif()

add_executable(p3cg src/main.cpp src/grid.cpp src/halo.cpp src/kernels.cpp src/cg.cpp src/record.cpp)
target_link_libraries(p3cg PRIVATE MPI::MPI_CXX)
target_compile_definitions(p3cg PRIVATE PLANT_HALO_BUG=${PLANT_HALO_BUG} P3_GIT_SHA="${P3_GIT_SHA}")
target_compile_options(p3cg PRIVATE -Wall -Wextra)
if(P3_NATIVE)
  target_compile_options(p3cg PRIVATE -march=native)
endif()
if(P3_OFFLOAD)
  separate_arguments(_off UNIX_COMMAND "${P3_OFFLOAD_FLAGS}")
  target_compile_definitions(p3cg PRIVATE P3_OFFLOAD=1)
  target_compile_options(p3cg PRIVATE ${_off})
  target_link_options(p3cg PRIVATE ${_off})
elseif(P3_OPENMP)
  target_link_libraries(p3cg PRIVATE OpenMP::OpenMP_CXX)
endif()

# The graph extension (CPU only): p3graph (community labelling with the same CG solvers, cg_impl.hpp, on the
# distributed CSR operator) and p3convert (SNAP edge list -> binary CSR; serial).
if(NOT P3_OFFLOAD)
  add_executable(p3graph src/graph/graph_main.cpp src/graph/dcsr.cpp src/graph/label.cpp src/kernels.cpp
                         src/grid.cpp src/record.cpp)
  target_link_libraries(p3graph PRIVATE MPI::MPI_CXX)
  target_compile_definitions(p3graph PRIVATE PLANT_GHOST_BUG=${PLANT_GHOST_BUG} P3_GIT_SHA="${P3_GIT_SHA}")
  target_compile_options(p3graph PRIVATE -Wall -Wextra)
  if(P3_NATIVE)
    target_compile_options(p3graph PRIVATE -march=native)
  endif()
  if(P3_OPENMP)
    target_link_libraries(p3graph PRIVATE OpenMP::OpenMP_CXX)
  endif()
  add_executable(p3convert src/graph/convert.cpp)
  target_compile_options(p3convert PRIVATE -Wall -Wextra)
endif()
if(NOT PLANT_GHOST_BUG STREQUAL "0")
  message(WARNING "PLANT_GHOST_BUG=${PLANT_GHOST_BUG}: p3graph is deliberately WRONG (for qualifying validate_graph.py)")
endif()

if(NOT PLANT_HALO_BUG STREQUAL "0")
  message(WARNING "PLANT_HALO_BUG=${PLANT_HALO_BUG}: this binary is deliberately WRONG (for qualifying validate.py)")
endif()
